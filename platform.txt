# Toolchain lookup: empty compiler.path means PATH lookup. For a pinned toolchain pass
#   --build-property "compiler.path=/path/to/xpack-riscv-none-elf-gcc/bin/"
compiler.path=
compiler.c.cmd=riscv-none-elf-gcc
compiler.cpp.cmd=riscv-none-elf-g++
compiler.ar.cmd=riscv-none-elf-gcc-ar
compiler.objcopy.cmd=riscv-none-elf-objcopy
compiler.size.cmd=riscv-none-elf-size

compiler.warning_flags=-Wall
# Link-time optimisation: 9-24% less flash (core_contracts, usb_classes,
# OepProbe, a TLS client), the same speed (CryptoBench on CH32X035: ECC and
# SHA-256 within 2%, AES-GCM 3.5% slower; 2026-10-21). The objects are fat -
# machine code beside the IR - because libc refers to _sbrk only after the
# LTO pass has run: with IR alone the core's _sbrk is not seen, libgloss's
# semihosting one is linked instead, and its ecall traps at the first malloc.
# -msave-restore takes 4-5% more but costs ECC 7-9% in speed, so it is a
# choice in the Optimize menu (opt) in boards.txt, as is building without LTO.
# With LTO the code is generated again at the link, so the combine recipe
# repeats the code-generation flags: -Os and the per-function sections that
# --gc-sections needs (on OepProbe for CH32X035, 48 bytes; it depends on the
# sketch).
build.optimize_flags=-flto -ffat-lto-objects
compiler.common_flags=-march={build.march} -mabi={build.mabi} -Os -g -ffunction-sections -fdata-sections {compiler.warning_flags} {build.optimize_flags}
# api/deprecated-avr-comp is on the include path so that <avr/dtostrf.h> and
# <avr/pgmspace.h> resolve the way AVR sketches spell them. ArduinoCore-API
# ships those headers for exactly that reason, and samd/renesas add the same
# path; without it the headers are present but unreachable.
# api/deprecated holds the pre-ArduinoCore-API header names (Print.h, Stream.h,
# WString.h, ...) as one-line forwards to api/. Libraries still include them
# bare - Adafruit_GFX does `#include "Print.h"` - and renesas adds the same path.
# Per-peripheral menu choices (USARTn: AUTO / Disable / FIX). A board with the
# menus sets this from them; every other board compiles with all on AUTO.
build.periph_defines=
# CH32H41x's VIO18 Supply menu (boards.txt); empty: the XO strap's voltage.
build.vio18_defines=
# The Clock Source menu (boards.txt): the crystal's setting on; empty: HSI.
build.clock_src_defines=
# The Flash / RAM split menu (boards.txt) on the parts that have it (CH32V303/V307/V317 RC, VC, WC, SC);
# empty: the linker script's split is the chip's own and nothing is checked (ch32rv_option_user.c).
build.ramsplit_defines=
# The Reset Pin menu's USER value and mask (ch32rv_option_user.c); empty where the series has no such option.
build.rstpin_defines=
compiler.c.flags={compiler.common_flags} -std=gnu11 {build.core_defines} {build.clock_defines} {build.clock_src_defines} {build.periph_defines} {build.vio18_defines} {build.ramsplit_defines} {build.rstpin_defines} {build.fp_flags} {compiler.stdio_flags} {compiler.vector_defines} "-I{build.core.path}/api/deprecated-avr-comp"
compiler.cpp.flags={compiler.common_flags} -std=gnu++17 {build.core_defines} {build.clock_defines} {build.clock_src_defines} {build.periph_defines} {build.vio18_defines} {build.ramsplit_defines} {build.rstpin_defines} {build.fp_flags} {compiler.stdio_flags} {compiler.vector_defines} -fno-exceptions -fno-rtti -fno-threadsafe-statics "-I{build.core.path}/api/deprecated-avr-comp" "-I{build.core.path}/api/deprecated"
compiler.S.flags={compiler.common_flags} -x assembler-with-cpp {build.startup_defines} {build.ramsplit_defines} {build.rstpin_defines} -DCH32RV_VECTORS=vectors_{build.vector_variant}.inc
# The three generated interrupt tables share one stem, so a board - or a
# single pnum menu entry, for a part on a different die - selects all
# three at once (CH32V203RBT6 is CH32V20x_D8 while the rest of CH32V203
# is D6, and their slot numbering diverges).
compiler.vector_defines=-DCH32RV_IRQNS=irqn_{build.vector_variant}.h -DCH32RV_EXTIS=exti_{build.vector_variant}.h -DCH32RV_CLOCK_INIT={build.clock_init}
# ARDUINO_{build.board} names the board and stays the same whichever part
# number the menu picks, as on arduino-esp32 (its chip-variant menu does not
# change ARDUINO_ESP32_DEV either). The series is CH32RV_SERIES_{build.series}
# (CH32RV_SERIES_CH32V003) - prefixed, since a bare CH32V003 is what other
# cores and the vendor SDKs test - and the part CH32RV_PART_{build.part}:
# CH32RV_PART_CH32V003F4U6, or CH32RV_PART_ANY for the ANY entry.
compiler.defines=-DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -DCH32RV_SERIES_{build.series} -DCH32RV_PART_{build.part} {compiler.version_defines}
# The platform version as numbers, for `#if CH32RV_VERSION >= CH32RV_VERSION_VAL(0, 1, 0)`
# (cores/arduino/ch32rv_version.h). Written from `version=` above by
# tools/index/gen_index.py, which the release workflow runs before it commits
# platform.txt back; tests/unit/test_version_macros.py keeps the two equal.
compiler.version_defines=-DCH32RV_VERSION_MAJOR=0 -DCH32RV_VERSION_MINOR=4 -DCH32RV_VERSION_PATCH=0

build.extra_flags=
# Chosen by the Floating Point menu (rtlib) in boards.txt. fp_flags reach every
# C/C++ compile: float or double arithmetic (CH32RV_FP_FLOAT with
# -fsingle-precision-constant), printf's %f (CH32RV_PRINTF_FLOAT), newlib's
# printf (CH32RV_PRINTF_NEWLIB); see cores/arduino/ch32rv_format.h. rtlib_flags
# reach the link: newlib-nano's float printf / scanf.
build.fp_flags=
build.rtlib_flags=
# The core defines printf, vprintf, sprintf, snprintf, vsprintf, vsnprintf,
# puts, putchar and strftime on its own formatter (ch32rv_stdio_string.c,
# ch32rv_stdio_out.c, ch32rv_strftime.c). As builtins, calls to them do not show in the LTO
# symbol table, so the core's definitions were pulled in only after LTO had
# dropped the formatter they call (an undefined ch32rv_vformat). Plain calls
# resolve during LTO. Costs GCC's rewrites such as printf("x\n") -> puts().
compiler.stdio_flags=-fno-builtin-printf -fno-builtin-vprintf -fno-builtin-sprintf -fno-builtin-snprintf -fno-builtin-vsprintf -fno-builtin-vsnprintf -fno-builtin-puts -fno-builtin-putchar -fno-builtin-strftime

# compiler.<lang>.extra_flags is where boards.local.txt or --build-property put
# per-user flags. A recipe that leaves them out silently ignores them, so every
# recipe references them. Distinct from build.extra_flags, which ADR-0007
# reserves for the user and the core never writes.
compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=
compiler.c.elf.extra_flags=

# A sketch's build_opt.h: compiler options, one or more per line, next to the
# .ino, applied to every C, C++ and assembler file of the build - sketch, core
# and libraries alike (ADR-0007). arduino-cli has no such feature of its own;
# the hooks below copy the file into the build directory (an empty one when the
# sketch has none) and the compile recipes read it with gcc's @file. A property
# of its own rather than compiler.<lang>.extra_flags, so that a user who sets
# those from --build-property keeps build_opt.h too.
build.opt.name=build_opt.h
build.opt.path={build.path}/{build.opt.name}
build.opt.flags="@{build.opt.path}"
recipe.hooks.prebuild.1.pattern=/usr/bin/env bash -c "[ ! -f '{build.source.path}/{build.opt.name}' ] || cp -f '{build.source.path}/{build.opt.name}' '{build.opt.path}'"
recipe.hooks.prebuild.2.pattern=/usr/bin/env bash -c "[ -f '{build.opt.path}' ] || : > '{build.opt.path}'"
recipe.hooks.prebuild.1.pattern.windows=cmd /c if exist "{build.source.path}\{build.opt.name}" COPY /y "{build.source.path}\{build.opt.name}" "{build.opt.path}"
recipe.hooks.prebuild.2.pattern.windows=cmd /c if not exist "{build.opt.path}" type nul > "{build.opt.path}"

recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {compiler.c.extra_flags} {compiler.defines} {build.extra_flags} {build.opt.flags} {includes} -c "{source_file}" -o "{object_file}"
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {compiler.cpp.extra_flags} {compiler.defines} {build.extra_flags} {build.opt.flags} {includes} -c "{source_file}" -o "{object_file}"
recipe.S.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.S.flags} {compiler.S.extra_flags} {compiler.defines} {build.extra_flags} {build.opt.flags} {includes} -c "{source_file}" -o "{object_file}"
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" rcs "{archive_file_path}" "{object_file}"
# newlib-nano is the default runtime; build.rtlib_flags selects float printf/scanf.
# ADR-0004 proposes newlib-nano as the default runtime with %f as a menu opt-in,
# but that ADR is still Proposed - it is not authority for this change. What is
# measured: the full newlib formatter is ~40 KB, so a printf sketch on CH32X035
# goes 48,492 -> 7,064 bytes and fits CH32V003's 16 KB at all. nano drops %f by
# default; the printf menu in boards.txt adds it back with -u _printf_float.
# --start-group around core.a and -lc: the core's syscall stubs must beat the
# libgloss ones, which are *semihosting* - they issue `ecall`, and a CH32 with
# no environment-call handler traps that straight to the reset vector. libgloss
# comes after core.a on the driver's implicit link line, so left-to-right
# resolution used to pick its _sbrk and _write: any sketch that allocated
# (String, new, malloc) or called printf() rebooted in a loop and printed
# nothing at all. The group makes the linker come back to core.a when libc asks
# for those symbols, and costs nothing when the sketch needs neither
# (Blink stays 932 bytes on CH32X035). The sketch's objects and the libraries'
# archives (dot_a_linkage) are in the group too: the archives come in the order
# the sketch includes them, and one library's archive may need another's that
# came earlier (a sketch with NetworkClientSecure.h before Network.h).
# --no-warn-rwx-segments: .init_array (writable) shares the single FLASH segment with
# text; acceptable for MCU flash layout. Revisit with PHDRS in the real core.
recipe.c.combine.pattern="{compiler.path}{compiler.c.cmd}" -march={build.march} -mabi={build.mabi} -Os -ffunction-sections -fdata-sections {build.optimize_flags} -nostartfiles --specs=nano.specs {build.rtlib_flags} {compiler.c.elf.extra_flags} -Wl,--gc-sections -Wl,--no-warn-rwx-segments "-L{build.core.path}" "-L{build.variant.path}" "-T{build.variant.path}/{build.ldscript}" "-Wl,-Map,{build.path}/{build.project_name}.map" -o "{build.path}/{build.project_name}.elf" -Wl,--start-group {object_files} "{build.path}/{archive_file}" -lc -lm -Wl,--end-group
recipe.objcopy.hex.pattern="{compiler.path}{compiler.objcopy.cmd}" -O ihex "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.hex"
recipe.objcopy.bin.pattern="{compiler.path}{compiler.objcopy.cmd}" -O binary "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"

recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.elf"
# .reset / .image: CH32RV_LOADCODE (sections_loadcode.ld) - the image is stored
# in flash and runs from RAM, so it counts on both sides.
recipe.size.regex=^(?:\.init|\.vector|\.text|\.rodata|\.data|\.init_array|\.preinit_array|\.fini_array|\.reset|\.image)\s+([0-9]+).*
recipe.size.regex.data=^(?:\.data|\.bss|\.image)\s+([0-9]+).*

# ---------------------------------------------------------------- upload
# ch32rv (ADR-0008). `flash --reset run --confirm-run` writes the image, starts
# the target and checks that it is actually running, which is what a programmer
# pattern has room for. The platform ships one uploader, and it is this one.
#
# {build.ch32rv_chip} is generated per menu entry from
# tools/index/ch32rv_chips.csv: the exact part where ch32rv knows it, the
# family name otherwise, and the series name for a series ch32rv has no name
# for - such a board is listed like any other (docs/support-status.ja.md says
# which series are flashable), and an upload attempted anyway stops with
# ch32rv's `target-not-in-db` (exit 20) before writing. Never `auto` here: that
# detects whatever is attached and writes to it. See ch32rv_chip() in
# tools/generate/generate.py.
tools.ch32rv.path={runtime.tools.ch32rv.path}
tools.ch32rv.cmd=ch32rv
tools.ch32rv.cmd.windows=ch32rv.exe

# Uploading with the IDE port alone (docs/oep-workflow.ja.md §3.2). Pick a port
# and press Upload: boards.txt gives every flashable board upload.protocol, so
# no programmer is asked for, and the tool follows the port's protocol:
#   serial   a probe's serial port - a WCH-Link's CDC, an OEP probe's USJ / CDC
#   wchlink  wchlink://<serial>, from ch32rv's discovery
#   oep      oep://<probe>/<port>, from ch32rv's discovery (<port> is the
#            name the probe's saved pin labels give, S of S.swdio / S.swio)
#   ch32rv-hid  ch32rv-hid://<topology>, the UIAPduino bootloader
#   ch32rv-app  ch32rv-app://<USB serial>, a running sketch with CH32RVUsbDevice's
#            DFU runtime (its interface named "CH32RV"); written over USB alone
#   ch32rv-isp  ch32rv-isp://<USB position>, the factory USB bootloader (1a86:55e0)
# ch32rv resolves the address itself, passed as it is. A port is chosen by its
# name, without attaching; a probe with several ports and none named is refused
# with the list (ch32rv 0.21, the reworked OEP spec: slots are gone).
# No plain `upload.tool`: arduino-cli takes it before the per-protocol keys
# below (measured, arduino-cli 1.5.1), so a ch32rv-app port would be handed to
# the debug-probe recipe. `upload.tool.default` covers an upload with no port.
upload.tool.default=ch32rv
upload.tool.serial=ch32rv
upload.tool.wchlink=ch32rv
upload.tool.oep=ch32rv
upload.tool.ch32rv-hid=ch32rv_hid
upload.tool.ch32rv-app=ch32rv_isp
upload.tool.ch32rv-isp=ch32rv_isp

# --progress none because a recipe's stdout is not a tty; ch32rv defaults to a
# progress bar only on one, but saying so keeps the log identical either way.
# --confirm-run fails (exit 50) when the target does not run after the reset.
# upload.erase_args comes from the Erase Flash menu: empty keeps the flash
# tail a sketch declared with CH32RV_FLASH_STORE (ch32rv's default,
# keep-store), "--erase chip" erases the whole chip. Every route is given it;
# one that cannot erase the chip (DFU, HID) refuses it with exit 24 rather
# than ignoring it.
upload.erase_args=
tools.ch32rv.upload.params.verbose=--verbose
tools.ch32rv.upload.params.quiet=
tools.ch32rv.upload.pattern="{path}/{cmd}" flash "{build.path}/{build.project_name}.elf" --format elf --chip {build.ch32rv_chip} --reset run --confirm-run --non-interactive --progress none {upload.verbose} {upload.erase_args} --probe "{upload.port.address}"

# `upload --programmer wch-link` (programmers.txt) stays for the probe-less
# command line: it takes the only WCH-Link attached, or the one named by
#   --upload-property upload.probe_args="--probe serial:<serial>"
# (`ch32rv probe list` prints the serial). Empty has to vanish from the line.
upload.probe_args=
tools.ch32rv.program.params.verbose=--verbose
tools.ch32rv.program.params.quiet=
tools.ch32rv.program.pattern="{path}/{cmd}" flash "{build.path}/{build.project_name}.elf" --format elf --chip {build.ch32rv_chip} --reset run --confirm-run --non-interactive --progress none {program.verbose} {upload.erase_args} {upload.probe_args}

# Over USB alone, through the factory USB ISP bootloader (docs/usb.ja.md). On a
# ch32rv-app port, --entry auto sends DFU_DETACH (or the 1200 bps touch) to the
# running sketch and waits for the bootloader; on a ch32rv-isp port the
# bootloader is already there and is written directly. --chip makes the ISP
# refuse a different part. The image is ISP protocol, not DFU download.
tools.ch32rv_isp.path={runtime.tools.ch32rv.path}
tools.ch32rv_isp.cmd=ch32rv
tools.ch32rv_isp.cmd.windows=ch32rv.exe
tools.ch32rv_isp.upload.params.verbose=--verbose
tools.ch32rv_isp.upload.params.quiet=
tools.ch32rv_isp.upload.pattern="{path}/{cmd}" --non-interactive --progress none isp flash "{build.path}/{build.project_name}.elf" --chip {build.ch32rv_chip} --entry auto {upload.erase_args} --device "{upload.port.address}" {upload.verbose}

# Over USB alone, on the parts whose sketch rewrites its own flash
# (CH32H41x, CH32RV_FLASH_SELF_WRITE): the running sketch's DFU interface
# takes DFU_DETACH, comes back in DFU mode and is written by DFU download from
# flash offset 0 (docs/usb.ja.md). Their factory ISP is out of reach of
# software, so boards.txt points their ch32rv-app ports here instead of at
# ch32rv_isp. `boot dfu` is in ch32rv 0.22.0 and later.
tools.ch32rv_dfu.path={runtime.tools.ch32rv.path}
tools.ch32rv_dfu.cmd=ch32rv
tools.ch32rv_dfu.cmd.windows=ch32rv.exe
tools.ch32rv_dfu.upload.params.verbose=--verbose
tools.ch32rv_dfu.upload.params.quiet=
tools.ch32rv_dfu.upload.pattern="{path}/{cmd}" --non-interactive --progress none boot dfu flash "{build.path}/{build.project_name}.bin" --format bin {upload.erase_args} --device "{upload.port.address}" {upload.verbose}

# UIAPduino Pro Micro CH32V003 V1.4 product bootloader. rv003usb takes the raw
# binary, and boot entry still needs an external action (docs/uiapduino-hid-upload.ja.md).
tools.ch32rv_hid.path={runtime.tools.ch32rv.path}
tools.ch32rv_hid.cmd=ch32rv
tools.ch32rv_hid.cmd.windows=ch32rv.exe
tools.ch32rv_hid.upload.params.verbose=--verbose
tools.ch32rv_hid.upload.params.quiet=
tools.ch32rv_hid.upload.pattern="{path}/{cmd}" --non-interactive --progress none boot hid flash "{build.path}/{build.project_name}.bin" {upload.erase_args} --device "{upload.port.address}" {upload.verbose}

# ---------------------------------------------------------------- discovery
# ch32rv lists what a plain serial scan cannot see or name: every WCH-Link as
# wchlink://<serial>, every labelled port of an OEP probe as oep://<probe>/<port> (a device
# with the OEP project USB ID 1209:4F45), and the UIAPduino
# bootloader as ch32rv-hid://, a running CH32RVUsbDevice sketch as ch32rv-app:// and
# the factory USB bootloader as ch32rv-isp://. It never opens an ordinary
# serial port. Declaring one
# discovery means the built-in ones have to be named too.
pluggable_discovery.required.0=builtin:serial-discovery
pluggable_discovery.required.1=builtin:mdns-discovery
pluggable_discovery.ch32rv.pattern="{runtime.tools.ch32rv.path}/ch32rv" arduino discovery

# ---------------------------------------------------------------- monitor
# Every monitor is ch32rv's (docs/oep-workflow.ja.md §6), for the IDE's serial
# ports too: its `source` setting reads the target's UART (default, a plain
# pass-through), SDI, DMDATA, DMSeq, RTT or an OEP probe's fixture UART, and its
# `chip` setting (boards.txt monitor_port.<protocol>.chip) keeps a debug-module
# source off a board that is not the one attached. On an ordinary serial port
# with no probe behind it the UART source is simply a serial monitor.
pluggable_monitor.pattern.serial="{runtime.tools.ch32rv.path}/ch32rv" arduino monitor --protocol serial
pluggable_monitor.pattern.wchlink="{runtime.tools.ch32rv.path}/ch32rv" arduino monitor --protocol wchlink
pluggable_monitor.pattern.oep="{runtime.tools.ch32rv.path}/ch32rv" arduino monitor --protocol oep
# ch32rv-app opens the sketch's own CDC port (none: the monitor says so);
# ch32rv-isp has nothing to show and refuses.
pluggable_monitor.pattern.ch32rv-app="{runtime.tools.ch32rv.path}/ch32rv" arduino monitor --protocol ch32rv-app
pluggable_monitor.pattern.ch32rv-isp="{runtime.tools.ch32rv.path}/ch32rv" arduino monitor --protocol ch32rv-isp

# ---------------------------------------------------------------- debug
# `arduino-cli debug` (and the IDE's debugger) drive GDB plus a GDB server.
# arduino-cli only understands `debug.server=openocd`, and it launches it as
#   openocd -s <scripts_dir> --file <script> -c "gdb_port pipe" -c "telnet_port 0"
# so the server speaks the GDB remote protocol on stdio, not a socket.
#
# **This platform does not ship OpenOCD.** A stock one cannot drive a WCH-Link
# at all - mainline has no `wlinke` adapter driver - and WCH does not publish
# the source of the build that can. So the binary is looked up on PATH, the way
# compiler.path already does it, and only the configuration is ours.
#
# Point it at one you already have (MounRiver Studio's, or the one another CH32
# core installed) without editing this file:
#   platform.local.txt:  debug.server.openocd.path=/path/to/openocd
#   arduino-cli debug --debug-property server.openocd.path=/path/to/openocd
# docs/debugger.ja.md has the per-OS locations and the rest of the recipe.
debug.executable={build.path}/{build.project_name}.elf
debug.toolchain=gcc
debug.toolchain.prefix=riscv-none-elf
debug.toolchain.path={runtime.tools.xpack-riscv-none-elf-gcc.path}/bin/
debug.server=openocd
debug.server.openocd.path=openocd
debug.server.openocd.scripts_dir={runtime.platform.path}/debug
debug.server.openocd.script={runtime.platform.path}/debug/ch32-riscv.cfg
