# ArduinoCore-CH32 platform. See docs/adr/README.ja.md (all ADRs currently Proposed).
name=CH32 RISC-V (ArduinoCore-CH32 prototype)
version=0.0.1

# Toolchain lookup: empty compiler.path means PATH lookup. For a pinned toolchain pass
#   --build-property "compiler.path=/path/to/xpack-riscv-none-elf-gcc/bin/"
compiler.path=
compiler.c.cmd=riscv-none-elf-gcc
compiler.cpp.cmd=riscv-none-elf-g++
compiler.ar.cmd=riscv-none-elf-gcc-ar
compiler.objcopy.cmd=riscv-none-elf-objcopy
compiler.size.cmd=riscv-none-elf-size

compiler.warning_flags=-Wall
compiler.common_flags=-march={build.march} -mabi={build.mabi} -Os -g -ffunction-sections -fdata-sections {compiler.warning_flags}
# api/deprecated-avr-comp is on the include path so that <avr/dtostrf.h> and
# <avr/pgmspace.h> resolve the way AVR sketches spell them. ArduinoCore-API
# ships those headers for exactly that reason, and samd/renesas add the same
# path; without it the headers are present but unreachable.
compiler.c.flags={compiler.common_flags} -std=gnu11 {build.core_defines} {build.clock_defines} {compiler.vector_defines} "-I{build.core.path}/api/deprecated-avr-comp"
compiler.cpp.flags={compiler.common_flags} -std=gnu++17 {build.core_defines} {build.clock_defines} {compiler.vector_defines} -fno-exceptions -fno-rtti -fno-threadsafe-statics "-I{build.core.path}/api/deprecated-avr-comp"
compiler.S.flags={compiler.common_flags} -x assembler-with-cpp {build.startup_defines} -DCH32_VECTORS=vectors_{build.vector_variant}.inc
# The three generated interrupt tables share one stem, so a board - or a
# single pnum menu entry, for a part on a different die - selects all
# three at once (CH32V203RBT6 is CH32V20x_D8 while the rest of CH32V203
# is D6, and their slot numbering diverges).
compiler.vector_defines=-DCH32_IRQNS=irqn_{build.vector_variant}.h -DCH32_EXTIS=exti_{build.vector_variant}.h -DCH32_CLOCK_INIT={build.clock_init}
compiler.defines=-DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -D{build.series}

build.extra_flags=
# Chosen by the printf menu in boards.txt; empty for the default entry.
build.printf_flags=

# compiler.<lang>.extra_flags is where arduino-cli injects a sketch's
# build_opt.h (as @<path>), and where boards.local.txt or --build-property put
# per-user flags. A recipe that leaves them out silently ignores both, so every
# recipe references them. Distinct from build.extra_flags, which ADR-0007
# reserves for the user and the core never writes.
compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=
compiler.c.elf.extra_flags=

recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} {compiler.c.extra_flags} {compiler.defines} {build.extra_flags} {includes} -c "{source_file}" -o "{object_file}"
recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} {compiler.cpp.extra_flags} {compiler.defines} {build.extra_flags} {includes} -c "{source_file}" -o "{object_file}"
recipe.S.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.S.flags} {compiler.S.extra_flags} {compiler.defines} {build.extra_flags} {includes} -c "{source_file}" -o "{object_file}"
recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" rcs "{archive_file_path}" "{object_file}"
# --specs=nano.specs: PROPOSED, NOT APPROVED (docs/approval-status.ja.md A-1).
# ADR-0004 proposes newlib-nano as the default runtime with %f as a menu opt-in,
# but that ADR is still Proposed - it is not authority for this change. What is
# measured: the full newlib formatter is ~40 KB, so a printf sketch on CH32X035
# goes 48,492 -> 7,064 bytes and fits CH32V003's 16 KB at all. nano drops %f by
# default; the printf menu in boards.txt adds it back with -u _printf_float.
# --start-group around core.a and -lc: the core's syscall stubs must beat the
# libgloss ones, which are *semihosting* - they issue `ecall`, and a CH32 with
# no environment-call handler traps that straight to the reset vector. libgloss
# comes after core.a on the driver's implicit link line, so left-to-right
# resolution used to pick its _sbrk and _write: any sketch that allocated
# (String, new, malloc) or called printf() rebooted in a loop and printed
# nothing at all. The group makes the linker come back to core.a when libc asks
# for those symbols, and costs nothing when the sketch needs neither
# (Blink stays 932 bytes on CH32X035).
# --no-warn-rwx-segments: .init_array (writable) shares the single FLASH segment with
# text; acceptable for MCU flash layout. Revisit with PHDRS in the real core.
recipe.c.combine.pattern="{compiler.path}{compiler.c.cmd}" -march={build.march} -mabi={build.mabi} -nostartfiles --specs=nano.specs {build.printf_flags} {compiler.c.elf.extra_flags} -Wl,--gc-sections -Wl,--no-warn-rwx-segments "-L{build.core.path}" "-L{build.variant.path}" "-T{build.variant.path}/{build.ldscript}" "-Wl,-Map,{build.path}/{build.project_name}.map" -o "{build.path}/{build.project_name}.elf" {object_files} -Wl,--start-group "{build.path}/{archive_file}" -lc -lm -Wl,--end-group
recipe.objcopy.hex.pattern="{compiler.path}{compiler.objcopy.cmd}" -O ihex "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.hex"
recipe.objcopy.bin.pattern="{compiler.path}{compiler.objcopy.cmd}" -O binary "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"

recipe.size.pattern="{compiler.path}{compiler.size.cmd}" -A "{build.path}/{build.project_name}.elf"
recipe.size.regex=^(?:\.init|\.vector|\.text|\.rodata|\.data|\.init_array|\.preinit_array|\.fini_array)\s+([0-9]+).*
recipe.size.regex.data=^(?:\.data|\.bss)\s+([0-9]+).*

# ---------------------------------------------------------------- upload
# probe-rs (ADR-0008). `download --reset` flashes and starts the target in one
# command, which is what a programmer pattern has room for.
#
# {build.probe_rs_chip} is generated per menu entry from
# tools/index/probe_rs_targets.csv; entries with no probe-rs target have no
# value and the board is labelled "[compile only]".
tools.probe_rs.path={runtime.tools.probe-rs.path}
tools.probe_rs.cmd=probe-rs
tools.probe_rs.cmd.windows=probe-rs.exe

# Probe selection. Empty means "the only probe attached"; with several probes
# pass the whole flag, because an empty {upload.probe_args} has to vanish from
# the command line rather than leave a dangling --probe:
#   --upload-property upload.probe_args="--probe 1a86:8010:FBC18F0680B0"
# `probe-rs list` prints that VID:PID:Serial form.
upload.probe_args=

# upload.tool must exist even though the documented path is
# `upload --programmer wch-link`: arduino-cli 1.3.1 resolves upload.tool.default
# before it looks at the programmer and fails with "Property
# 'upload.tool.default' is undefined" otherwise. A bare `upload` still refuses
# with "A programmer is required to upload", which is correct - there is no
# serial bootloader to fall back to.
upload.tool=probe_rs
upload.tool.default=probe_rs
tools.probe_rs.upload.params.verbose=
tools.probe_rs.upload.params.quiet=--disable-progressbars

tools.probe_rs.program.params.verbose=
tools.probe_rs.program.params.quiet=--disable-progressbars

tools.probe_rs.upload.pattern="{path}/{cmd}" download --chip {build.probe_rs_chip} --binary-format elf --non-interactive --reset {upload.verbose} {upload.probe_args} "{build.path}/{build.project_name}.elf"
tools.probe_rs.program.pattern="{path}/{cmd}" download --chip {build.probe_rs_chip} --binary-format elf --non-interactive --reset {program.verbose} {upload.probe_args} "{build.path}/{build.project_name}.elf"

# ---------------------------------------------------------------- monitor
# The UART Serial is an ordinary serial port, so the monitor arduino-cli ships
# is the right one; declaring it is what makes a Board Manager install pull it
# in rather than relying on the fallback. The per-board default baud rate is in
# boards.txt (monitor_port.serial.baudrate), which is where the specification
# puts port settings.
#
# The debug-module channels are NOT wired here. SerialSDI needs none: the probe
# forwards it to its own USB CDC port, so the serial monitor reads it once the
# probe has been told to (`wlink sdi-print enable`). SerialRTT and SerialDMDATA
# would each need a pluggable monitor tool of our own - see docs/todo.ja.md.
pluggable_monitor.required.serial=builtin:serial-monitor
