# Upload back ends. CH32 flashing is SWD, not serial, so these are `programmer`
# entries driving tools.<t>.program.pattern - `arduino-cli upload --programmer
# <id>` needs no --port, and keeps using this path even when the pytest plugin
# passes --port for the runtime serial (experiment 0009).
#
# Selecting one probe out of several: pass its USB serial through
#   arduino-cli upload --programmer wch-link --upload-property upload.probe_args="--probe serial:<serial>"
# `ch32rv probe list` prints the serial. Every WCH-Link reports a unique one
# (experiment 0011), so this is deterministic.
#
# The id stays `wch-link` across the move from probe-rs to ch32rv (ADR-0008):
# it names the probe, not the tool, so every generated sketch.yaml keeps
# working.

wch-link.name=WCH-Link / WCH-LinkE (ch32rv)
wch-link.program.tool=ch32rv
wch-link.program.tool.default=ch32rv
